# -------------------------------------------------------
# Default paths for third-party libraries
# -------------------------------------------------------

if(WIN32)

   # Specify the Visual Studio version used to build the dependent libraries.
   # We maintain two sets of prebuilt libraries compiled with Visual Studio 2017
   # and Visual Studio 2022. Make sure the selected version matches the compiler
   # used to build Tobasa.
   #
   # Supported values: vs2022, vs2017
   set(MSVS_VER "vs2022")

   # Tobasa Library build as static library, CRT Linkage: static,  Deps Library Linkage: static
   set(MSVS_LINK "static")

   if(TOBASA_BUILD_MONOLITHIC_SHARED_LIB)
      # One Tobasa DLL, dynamic CRT, and static third-party libraries.
      set(MSVS_LINK "tbs-shared-monolithic")
   elseif(TOBASA_BUILD_SHARED_LIB)
      # Tobasa Library build as shared library, CRT Linkage: dynamic, Deps Library Linkage: static
      set(MSVS_LINK "tbs-shared")
   endif()

   if(TOBASA_BUILD_WITH_VCPKG)         
      set(VCPKG_TARGET_TRIPLET_DEF     "x64-windows-${MSVS_LINK}-${MSVS_VER}" )
      set(VCPKG_OVERLAY_PORTS_DEF      "d:/vcpkg/my_ports_${MSVS_VER}" )
      set(VCPKG_TOOLCHAIN_FILE_DEF     "d:/vcpkg/scripts/buildsystems/vcpkg.cmake" )
   endif() 

   set(TOBASA_OPENSSL_DIR_DEF          "d:/dev/tbs_3rdparty_packages/openssl_x64-windows-${MSVS_LINK}-${MSVS_VER}" )
   set(TOBASA_LIBPQ_DIR_DEF            "d:/dev/tbs_3rdparty_packages/libpq_x64-windows-${MSVS_LINK}-${MSVS_VER}" )
   
   set(TOBASA_ZLIB_DIR_DEF             "d:/dev/tbs_3rdparty_packages/zlib_x64-windows-${MSVS_LINK}-${MSVS_VER}" )
   set(TOBASA_MINIZIP_DIR_DEF          "d:/dev/tbs_3rdparty_packages/minizip_x64-windows-${MSVS_LINK}-${MSVS_VER}" )

   set(TOBASA_PROTOBUF_DIR_DEF         "d:/dev/tbs_3rdparty_packages/protobuf_x64-windows-${MSVS_LINK}-${MSVS_VER}" )
   set(TOBASA_HTTP2_DIR_DEF            "d:/dev/tbs_3rdparty_packages/nghttp2_x64-windows-${MSVS_LINK}-${MSVS_VER}" )

   set(TOBASA_LIBPNG_DIR_DEF           "d:/dev/tbs_3rdparty_packages/libpng_x64-windows-${MSVS_LINK}-${MSVS_VER}" )
   set(TOBASA_LIBXML2_DIR_DEF          "d:/dev/tbs_3rdparty_packages/libxml2_x64-windows-${MSVS_LINK}-${MSVS_VER}" )
   set(TOBASA_OPENJPEG_DIR_DEF         "d:/dev/tbs_3rdparty_packages/openjpeg_x64-windows-${MSVS_LINK}-${MSVS_VER}" )
   set(TOBASA_LIBTIFF_DIR_DEF          "d:/dev/tbs_3rdparty_packages/tiff_x64-windows-${MSVS_LINK}-${MSVS_VER}" )

else()
   set(TOBASA_PROTOBUF_DIR_DEF         "/opt/protobuf" )
   set(TOBASA_HTTP2_DIR_DEF            "/opt/nghttpd" )
endif()
